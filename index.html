// Initialize Supabase Client const SUPABASE_URL = "YOUR_SUPABASE_URL"; const SUPABASE_ANON_KEY = "YOUR_SUPABASE_ANON_KEY"; const supabase = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY); let currentUserProfile = null; // Initialize app data on page load document.addEventListener('DOMContentLoaded', async () => { await fetchDynamicSettings(); await initializeAuthSession(); await loadMembersData(); }); // Fetch dynamic settings from Supabase async function fetchDynamicSettings() { const { data: settings, error } = await supabase.from('app_settings').select('*'); if (error) return console.error('Error fetching settings:', error); settings.forEach(setting => { if (setting.key === 'app_title') { const titleElem = document.getElementById('app-title') || document.querySelector('header h1'); if (titleElem) titleElem.innerText = setting.value; } if (setting.key === 'portal_notice') { const noticeElem = document.getElementById('portal-notice'); if (noticeElem) noticeElem.innerText = setting.value; } }); } // Manage user session & evaluate role permissions dynamically async function initializeAuthSession() { const { data: { session } } = await supabase.auth.getSession(); if (!session) return; // Retrieve full profile and dynamic role assignment from database const { data: profile, error } = await supabase .from('profiles') .select('*') .eq('id', session.user.id) .single(); if (error || !profile) return; currentUserProfile = profile; // Dynamic UI role evaluation applyRoleBasedUI(profile.role); } // Dynamic display handling according to database role function applyRoleBasedUI(role) { const adminElements = document.querySelectorAll('.admin-only, [data-role="admin"]'); const officialElements = document.querySelectorAll('.official-only, [data-role="official"]'); adminElements.forEach(el => el.style.display = (role === 'admin') ? '' : 'none'); officialElements.forEach(el => el.style.display = (role === 'admin' || role === 'official') ? '' : 'none'); } // Fetch member table entries dynamically from database async function loadMembersData() { const tableBody = document.getElementById('members-table-body') || document.querySelector('tbody'); if (!tableBody) return; const { data: members, error } = await supabase .from('members') .select('*') .order('created_at', { ascending: false }); if (error) { console.error("Error loading members:", error); return; } tableBody.innerHTML = ''; members.forEach(member => { const tr = document.createElement('tr'); tr.innerHTML = ` ${member.membership_number} ${member.full_name} ${member.email} ${member.phone || 'N/A'} ${member.status} `; tableBody.appendChild(tr); }); // Re-apply role visibility to dynamically added action buttons if (currentUserProfile) { applyRoleBasedUI(currentUserProfile.role); } } // Dynamically insert member record (replaces local array pushes) async function handleAddMember(event) { if (event) event.preventDefault(); const name = document.getElementById('member-name').value; const email = document.getElementById('member-email').value; const phone = document.getElementById('member-phone').value; const memNo = document.getElementById('member-no').value; const { data, error } = await supabase .from('members') .insert([{ full_name: name, email: email, phone: phone, membership_number: memNo }]); if (error) { alert("Failed to create member: " + error.message); } else { await loadMembersData(); // Close modal using existing modal function if present if (typeof closeModal === 'function') closeModal(); } } // Dynamically remove member record async function deleteMember(id) { if (!confirm("Are you sure you want to delete this record?")) return; const { error } = await supabase .from('members') .delete() .eq('id', id); if (error) { alert("Failed to delete member: " + error.message); } else { await loadMembersData(); } }
