-- STEP 4: Now create functions AFTER policies DROP FUNCTION IF EXISTS get_dynamic_dashboard(); CREATE OR REPLACE FUNCTION get_dynamic_dashboard() RETURNS JSON AS $$ DECLARE result JSON; BEGIN SELECT json_build_object( 'settings', (SELECT COALESCE(json_agg(row_to_json(s)), '[]'::json) FROM (SELECT key, value FROM app_settings) s), 'profile', (SELECT row_to_json(p) FROM (SELECT id, role FROM profiles WHERE id = auth.uid()) p), 'members', (SELECT COALESCE(json_agg(row_to_json(m)), '[]'::json) FROM (SELECT * FROM members ORDER BY created_at DESC LIMIT 100) m) ) INTO result; RETURN result; END; $$ LANGUAGE plpgsql SECURITY DEFINER; DROP FUNCTION IF EXISTS create_member_dynamic(TEXT,TEXT,TEXT,TEXT); CREATE OR REPLACE FUNCTION create_member_dynamic( p_full_name TEXT, p_email TEXT, p_phone TEXT, p_membership_number TEXT ) RETURNS JSON AS $$ DECLARE new_member JSON; BEGIN INSERT INTO members (full_name, email, phone, membership_number, user_id) VALUES (p_full_name, p_email, p_phone, p_membership_number, auth.uid()) RETURNING row_to_json(members.*) INTO new_member; RETURN new_member; END; $$ LANGUAGE plpgsql SECURITY DEFINER;
